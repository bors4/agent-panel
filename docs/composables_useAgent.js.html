/**
 * Composable для управления AI агентом (Vue 3).
 */

import { ref, computed } from "vue";
import { useToast } from "./useToast.js";
import {
  runAgentLoop,
  executeTool,
  getSession,
  addToSession,
} from "../api/client.js";

/**
 * Composable для работы с AI агентом.
 * @returns {Object} Объект с методами и вычисляемыми свойствами
 */
export function useAgent() {
  const isProcessing = ref(false);
  const currentChatId = ref(null);
  const pendingToolCall = ref(null);

  const toast = useToast();

  /**
   * Запустить цикл агента.
   * @param {Object} agentState - Состояние агента
   * @returns {Promise<Object>} Результат выполнения цикла
   */
  async function runLoop(agentState) {
    if (isProcessing.value) return;

    isProcessing.value = true;
    pendingToolCall.value = null;

    try {
      const result = await runAgentLoop(agentState);

      if (result.type === "await_tool") {
        pendingToolCall.value = result.toolCall;
        toast.info("Waiting for tool execution...");
        return result;
      } else if (result.type === "complete") {
        toast.success(result.response);
        return result;
      } else if (result.type === "error") {
        toast.error(result.message);
        throw new Error(result.message);
      }

      return result;
    } catch (error) {
      console.error("Agent loop error:", error);
      toast.error(error.message || "Unknown error");
      throw error;
    } finally {
      isProcessing.value = false;
    }
  }

  /**
   * Выполнить инструмент, который ожидает выполнения.
   * @param {Object} toolCall - Вызов инструмента
   * @returns {Promise<Object>} Результат выполнения инструмента
   */
  async function executePendingTool(toolCall) {
    if (!pendingToolCall.value || pendingToolCall.value.name !== toolCall.name) {
      throw new Error("No pending tool to execute");
    }

    const result = await executeTool({ ...toolCall, args: pendingToolCall.value.args });

    if (result.success) {
      toast.success(`Tool ${toolCall.name} executed successfully`);
    } else {
      toast.error(`Tool ${toolCall.name} failed: ${result.error}`);
    }

    return result;
  }

  /**
   * Добавить сообщение в текущую сессию.
   * @param {'user'|'assistant'} role - Роль отправителя
   * @param {string} content - Контент сообщения
   */
  async function addMessage(role, content) {
    if (!currentChatId.value) return;

    await addToSession(currentChatId.value, role, content);
  }

  /**
   * Создать новую сессию чата.
   * @returns {Promise<number>} ID новой сессии
   */
  async function createNewSession() {
    const chatId = Date.now().toString();
    currentChatId.value = chatId;
    return chatId;
  }

  /**
   * Получить текущую сессию.
   * @returns {Object|null} Объект сессии или null
   */
  function getCurrentSession() {
    if (!currentChatId.value) return null;
    return getSession(currentChatId.value);
  }

  return {
    isProcessing,
    currentChatId,
    pendingToolCall,
    runLoop,
    executePendingTool,
    addMessage,
    createNewSession,
    getCurrentSession,
  };
}
