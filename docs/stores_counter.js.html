/**
 * Pinia store для управления счетчиком (пример).
 */

import { defineStore } from "pinia";
import { ref, computed } from "vue";

const STORAGE_KEY = "counter-store";

/**
 * Store для глобального состояния счетчика.
 * @returns {Object} Методы и вычисляемые свойства store
 */
export const useCounterStore = defineStore("counter", () => {
  const count = ref(0);
  const isInitialized = ref(false);

  /**
   * Увеличить значение счетчика.
   * @param {number} increment - Прирост значения (по умолчанию 1)
   */
  function increment(increment = 1) {
    count.value += increment;
  }

  /**
   * Уменьшить значение счетчика.
   * @param {number} decrement - Сброс значения (по умолчанию 1)
   */
  function decrement(decrement = 1) {
    count.value -= decrement;
  }

  /**
   * Установить новое значение счетчика.
   * @param {number} value - Новое значение
   */
  function setCount(value) {
    count.value = value;
  }

  /**
   * Сбросить счетчик в ноль.
   */
  function reset() {
    count.value = 0;
  }

  /**
   * Загрузить состояние из localStorage.
   */
  function loadFromStorage() {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        const data = JSON.parse(stored);
        count.value = data.count || 0;
        isInitialized.value = true;
      }
    } catch (error) {
      console.warn("Failed to load from storage:", error);
    }
  }

  /**
   * Сохранить состояние в localStorage.
   */
  function saveToStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify({ count: count.value }));
    } catch (error) {
      console.warn("Failed to save to storage:", error);
    }
  }

  /**
   * Вычисляемое свойство: отрицательное значение.
   */
  const negativeCount = computed(() => -count.value);

  return {
    count,
    isInitialized,
    increment,
    decrement,
    setCount,
    reset,
    loadFromStorage,
    saveToStorage,
    negativeCount,
  };
});
