/**
 * API клиент для взаимодействия с backend сервером.
 */

const BASE_URL = "http://127.0.0.1:3000/api";
const API_KEY = import.meta.env.VITE_API_KEY || "agent-secret-key";

/**
 * Выполнить запрос к API с аутентификацией.
 * @param {string} endpoint - Endpoint API
 * @param {Object} options - Опции fetch (method, body, etc.)
 * @returns {Promise<Response>} Response от сервера
 */
async function apiFetch(endpoint, options = {}) {
  const url = `${BASE_URL}${endpoint}`;
  
  const defaultHeaders = {
    "Content-Type": "application/json",
    "x-api-key": API_KEY,
  };

  const headers = {
    ...defaultHeaders,
    ...options.headers,
  };

  try {
    const response = await fetch(url, {
      ...options,
      headers: options.body ? headers : defaultHeaders,
    });

    if (!response.ok) {
      throw new Error(`API error ${response.status}: ${response.statusText}`);
    }

    return response;
  } catch (error) {
    console.error("API request failed:", error);
    throw error;
  }
}

/**
 * Получить статус сервера.
 * @returns {Promise<Object>} Статус и статистика

export async function getStatus() {
  const response = await apiFetch("/status");
  return response.json();
}
